"""
Detect-process monitor using Frida. Console-only, zero disk I/O.

Blocks the target process from SEEING (open / query / enumerate):
  - C:\\ProgramData\\**
  - C:\\Users\\xkev5\\AppData\\Local\\Microsoft\\Windows\\Caches\\**
  - C:\\Program Files (x86)\\**
  - C:\\Windows\\Prefetch\\*.pf   (folder listing allowed)
  - Event Viewer: \\winevt\\Logs\\**, *.evtx, eventvwr.exe, eventvwr.msc
  - Autoruns: any path containing '\\autoruns', autoruns*.exe, autorunsc*.exe

Blocks registry:
  - Writes to ScriptBlockLogging / Transcript / AMSI / Defender watchlist
  - Reads or writes to \\autoruns* and \\Image File Execution Options\\

Logs registry access (reads + writes) to stdout. Writes nothing to disk.
"""

import sys
import time
import ctypes
import threading

import frida
import psutil

# ------------------------------------------------------------------
# Configuration
# ------------------------------------------------------------------
TARGET_SUBSTRING      = "detect"
DEDUP_WINDOW_SECONDS  = 60
POLL_INTERVAL         = 1.0
PERSIST_INTERVAL      = 30.0
# ------------------------------------------------------------------

_dedup: dict[tuple, float] = {}
_dedup_lock = threading.Lock()


def out(msg: str):
    try:
        sys.stdout.write(msg + "\n")
        sys.stdout.flush()
    except Exception:
        pass


def log_dedup(key: tuple, message: str):
    now = time.time()
    with _dedup_lock:
        last = _dedup.get(key, 0)
        if now - last < DEDUP_WINDOW_SECONDS:
            return
        _dedup[key] = now
        if len(_dedup) > 5000:
            cutoff = now - DEDUP_WINDOW_SECONDS
            for k in [k for k, t in _dedup.items() if t < cutoff]:
                _dedup.pop(k, None)
    out(message)


def is_admin() -> bool:
    try:
        return ctypes.windll.shell32.IsUserAnAdmin() != 0
    except Exception:
        return False


def require_admin():
    if not is_admin():
        sys.stderr.write("ERROR: run as Administrator.\n")
        sys.exit(1)


# ==================================================================
# Frida agent
# ==================================================================
AGENT_JS = r"""
'use strict';

const STATUS_ACCESS_DENIED = 0xC0000022;

// ---- NT -> DOS path map (built once, in the agent) ------------------------
const driveMap = {};
(function buildDriveMap() {
    const kernel32 = Process.getModuleByName('kernel32.dll');
    const GetLogicalDriveStringsW = new NativeFunction(
        kernel32.getExportByName('GetLogicalDriveStringsW'),
        'uint32', ['uint32', 'pointer']);
    const QueryDosDeviceW = new NativeFunction(
        kernel32.getExportByName('QueryDosDeviceW'),
        'uint32', ['pointer', 'pointer', 'uint32']);
    const buf = Memory.alloc(1024);
    const n = GetLogicalDriveStringsW(512, buf);
    if (n > 0) {
        const s = buf.readUtf16String(n);
        s.split('\x00').filter(Boolean).forEach(function (drive) {
            const letter = drive.replace(/\\+$/, '');
            const tgt = Memory.alloc(1024);
            if (QueryDosDeviceW(Memory.allocUtf16String(letter), tgt, 512) > 0) {
                driveMap[tgt.readUtf16String()] = letter;
            }
        });
    }
})();

function ntToDos(p) {
    if (!p) return p;
    for (const nt in driveMap) {
        if (p.startsWith(nt)) return driveMap[nt] + p.substring(nt.length);
    }
    return p;
}

// ---- protected-path rules -------------------------------------------------
function isProtected(dosPath) {
    if (!dosPath) return false;
    const p = dosPath.toLowerCase().replace(/\//g, '\\');

    // ProgramData (everything)
    if (p.startsWith('c:\\programdata\\')) return true;

    // User caches (specific user path)
    if (p.startsWith('c:\\users\\xkev5\\appdata\\local\\microsoft\\windows\\caches\\'))
        return true;

    // Program Files (x86) (everything)
    if (p.startsWith('c:\\program files (x86)\\' )) return true;

    // Autoruns (file names / paths)
    if (p.includes('\\autoruns')) return true;
    if (p.endsWith('autoruns.exe') ||
        p.endsWith('autorunsc.exe') ||
        p.endsWith('autoruns64.exe') ||
        p.endsWith('autorunsc64.exe')) return true;

    // Prefetch FILES only (folder listing allowed)
    if (p.startsWith('c:\\windows\\prefetch\\') && p.endsWith('.pf')) return true;

    // Event Viewer
    if (p.includes('\\winevt\\logs\\')) return true;
    if (p.endsWith('.evtx')) return true;
    if (p.endsWith('\\eventvwr.exe') || p.endsWith('\\eventvwr.msc')) return true;

    return false;
}

function isProtectedReg(keyPath) {
    if (!keyPath) return false;
    const k = keyPath.toLowerCase();
    if (k.includes('\\autoruns')) return true;
    if (k.includes('\\image file execution options\\')) return true;
    return false;
}

// Registry WRITE watchlist (value names)
const WATCHED_VALUE_NAMES = [
    'enablescriptblocklogging',
    'enabletranscripting',
    'enablemodulelog',
    'enableinvocationheader',
    'enablecmdlineauditing',
    'enablescriptblockinvocationlogging',
    'disableantispyware',
    'disablerealtimemonitoring',
    'disablebehaviormonitoring',
    'disableioavprotection',
    'disablescriptscanning'
];

// ---- helpers --------------------------------------------------------------
function readUnicodeString(p) {
    if (p.isNull()) return null;
    try {
        const len = p.readU16();
        if (len === 0 || len > 0x8000) return null;
        const buf = p.add(8).readPointer();
        if (buf.isNull()) return null;
        return buf.readUtf16String(len / 2);
    } catch (e) { return null; }
}

function readObjectAttributes(oaPtr) {
    if (oaPtr.isNull()) return null;
    try {
        return readUnicodeString(oaPtr.add(0x10).readPointer());
    } catch (e) { return null; }
}

const ntdll = Process.getModuleByName('ntdll.dll');
const NtQueryKey = new NativeFunction(
    ntdll.getExportByName('NtQueryKey'),
    'int', ['pointer', 'int', 'pointer', 'uint32', 'pointer']);

function keyHandleToPath(hKey) {
    if (hKey.isNull()) return null;
    const size = 0x2000;
    const buf = Memory.alloc(size);
    const ret = Memory.alloc(4);
    const st = NtQueryKey(hKey, 3, buf, size, ret);  // KeyNameInformation
    if (st !== 0) return null;
    const nlen = buf.readU32();
    if (nlen === 0 || nlen > 0x1FFE) return null;
    return buf.add(4).readUtf16String(nlen / 2);
}

// ---- file open hooks ------------------------------------------------------
function hookOpen(name) {
    const addr = Module.getExportByName('ntdll.dll', name);
    Interceptor.attach(addr, {
        onEnter(args) {
            this.blocked = false;
            const ntPath = readObjectAttributes(args[2]);
            if (!ntPath) return;
            const dos = ntToDos(ntPath);
            if (isProtected(dos)) {
                this.blocked = true;
                send({ type: 'block', op: name, path: dos });
            } else {
                send({ type: 'open', op: name, path: dos });
            }
        },
        onLeave(retval) {
            if (this.blocked) retval.replace(ptr(STATUS_ACCESS_DENIED));
        }
    });
}

// ---- registry open / create hooks ----------------------------------------
function hookRegistryOpen(name, keyArgIndex) {
    const addr = Module.getExportByName('ntdll.dll', name);
    Interceptor.attach(addr, {
        onEnter(args) {
            this.blocked = false;
            const ntPath = readObjectAttributes(args[keyArgIndex]);
            if (!ntPath) return;
            if (isProtectedReg(ntPath)) {
                this.blocked = true;
                send({ type: 'block', op: name, path: ntPath });
            } else {
                send({ type: 'reg', op: name, path: ntPath });
            }
        },
        onLeave(retval) {
            if (this.blocked) retval.replace(ptr(STATUS_ACCESS_DENIED));
        }
    });
}

// ---- registry WRITE hooks -------------------------------------------------
function hookSetValue() {
    const addr = Module.getExportByName('ntdll.dll', 'NtSetValueKey');
    Interceptor.attach(addr, {
        onEnter(args) {
            this.blocked = false;
            const hKey = args[0];
            const vname = readUnicodeString(args[1]);
            if (!vname) return;
            const keyPath = keyHandleToPath(hKey) || '(unknown key)';
            const v = vname.toLowerCase();
            const badValue = WATCHED_VALUE_NAMES.some(w => v === w || v.includes(w));
            const badKey = isProtectedReg(keyPath);
            if (badValue || badKey) {
                this.blocked = true;
                send({ type: 'block', op: 'NtSetValueKey',
                       path: keyPath + ' :: ' + vname });
            } else {
                send({ type: 'regwrite', op: 'NtSetValueKey',
                       path: keyPath + ' :: ' + vname });
            }
        },
        onLeave(retval) {
            if (this.blocked) retval.replace(ptr(STATUS_ACCESS_DENIED));
        }
    });
}

function hookDeleteValue() {
    const addr = Module.getExportByName('ntdll.dll', 'NtDeleteValueKey');
    Interceptor.attach(addr, {
        onEnter(args) {
            const keyPath = keyHandleToPath(args[0]) || '(unknown key)';
            const vname = readUnicodeString(args[1]);
            if (isProtectedReg(keyPath)) {
                send({ type: 'block', op: 'NtDeleteValueKey',
                       path: keyPath + ' :: ' + (vname || '') });
                this.blocked = true;
            } else {
                send({ type: 'regwrite', op: 'NtDeleteValueKey',
                       path: keyPath + ' :: ' + (vname || '') });
            }
        },
        onLeave(retval) {
            if (this.blocked) retval.replace(ptr(STATUS_ACCESS_DENIED));
        }
    });
}

// ---- install all hooks ----------------------------------------------------
['NtCreateFile', 'NtOpenFile',
 'NtQueryAttributesFile', 'NtQueryFullAttributesFile'].forEach(function (n) {
    try { hookOpen(n); } catch (e) { send({type:'err', msg: n + ': ' + e}); }
});

try { hookRegistryOpen('NtOpenKey', 0); } catch (e) {}
try { hookRegistryOpen('NtOpenKeyEx', 1); } catch (e) {}
try { hookRegistryOpen('NtCreateKey', 2); } catch (e) {}

try { hookSetValue(); } catch (e) {}
try { hookDeleteValue(); } catch (e) {}

send({ type: 'ready' });
"""


# ==================================================================
# Frida session management
# ==================================================================
class TargetMonitor:
    def __init__(self, pid: int, name: str):
        self.pid = pid
        self.name = name
        self.session = None
        self.script = None

    def start(self):
        try:
            self.session = frida.attach(self.pid)
        except frida.ProcessNotFoundError:
            return
        except Exception as e:
            out(f"[!] attach failed pid={self.pid}: {e}")
            return
        try:
            self.script = self.session.create_script(AGENT_JS)
            self.script.on("message", self._on_message)
            self.script.load()
        except Exception as e:
            out(f"[!] script load failed pid={self.pid}: {e}")

    def _on_message(self, message, data):
        if message.get("type") == "error":
            return
        payload = message.get("payload") or {}
        kind = payload.get("type")
        path = payload.get("path", "")
        op = payload.get("op", "")

        if kind == "ready":
            log_dedup((self.pid, "ready"),
                      f"[+] attached pid={self.pid} name={self.name}")
        elif kind == "block":
            log_dedup((self.pid, "block", path, op),
                      f"[BLOCK] pid={self.pid} {op} {path}")
        elif kind == "open":
            log_dedup((self.pid, "open", path, op),
                      f"[OPEN ] pid={self.pid} {op} {path}")
        elif kind == "reg":
            log_dedup((self.pid, "reg", path, op),
                      f"[REG  ] pid={self.pid} {op} {path}")
        elif kind == "regwrite":
            log_dedup((self.pid, "regwrite", path, op),
                      f"[REGW ] pid={self.pid} {op} {path}")

    def stop(self):
        try:
            if self.script:
                self.script.unload()
        except Exception:
            pass
        try:
            if self.session:
                self.session.detach()
        except Exception:
            pass
        self.script = None
        self.session = None


def find_targets():
    m = {}
    for p in psutil.process_iter(["pid", "name"]):
        try:
            if TARGET_SUBSTRING in (p.info["name"] or "").lower():
                m[p.pid] = p.info["name"]
        except (psutil.NoSuchProcess, psutil.AccessDenied):
            continue
    return m


# ==================================================================
# Registry persistence snapshotter (registry-only, no disk walks)
# ==================================================================
try:
    import winreg
    HAVE_WINREG = True
except ImportError:
    HAVE_WINREG = False

PERSIST_KEYS = [
    (r"SOFTWARE\Microsoft\Windows\CurrentVersion\Run", "Run"),
    (r"SOFTWARE\Microsoft\Windows\CurrentVersion\RunOnce", "RunOnce"),
    (r"SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\Run", "RunWOW"),
    (r"SOFTWARE\Wow6432Node\Microsoft\Windows\CurrentVersion\RunOnce", "RunOnceWOW"),
    (r"SYSTEM\CurrentControlSet\Services", "Services"),
    (r"SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon", "Winlogon"),
    (r"SOFTWARE\Microsoft\Windows NT\CurrentVersion\Image File Execution Options", "IFEO"),
]


def snapshot_registry():
    snap = {}
    if not HAVE_WINREG:
        return snap
    for key, label in PERSIST_KEYS:
        try:
            with winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, key,
                                0, winreg.KEY_READ) as k:
                i = 0
                while True:
                    try:
                        name, val, _ = winreg.EnumValue(k, i)
                        snap[f"{label}\\{name}"] = str(val)[:200]
                        i += 1
                    except OSError:
                        break
        except OSError:
            continue
    return snap


def persistence_loop():
    prev = None
    while True:
        try:
            cur = snapshot_registry()
            if prev is None:
                out(f"[PERSIST] baseline captured ({len(cur)} entries)")
            else:
                for k in cur.keys() - prev.keys():
                    out(f"[PERSIST-NEW] {k} -> {cur[k]}")
                for k in prev.keys() - cur.keys():
                    out(f"[PERSIST-DEL] {k}")
                for k in cur.keys() & prev.keys():
                    if cur[k] != prev[k]:
                        out(f"[PERSIST-MOD] {k}: {prev[k]} -> {cur[k]}")
            prev = cur
        except Exception as e:
            out(f"[!] persistence error: {e}")
        time.sleep(PERSIST_INTERVAL)


# ==================================================================
# Main loop
# ==================================================================
def main():
    require_admin()
    out("=" * 60)
    out("Detect-process Frida monitor v3 (console-only, zero disk I/O)")
    out("Blocks: ProgramData | Users\\xkev5\\...\\Caches | Program Files (x86)")
    out("        Prefetch *.pf | Event Viewer | Autoruns")
    out("Registry: ScriptBlockLogging + IFEO + autoruns writes blocked")
    out(f"dedup={DEDUP_WINDOW_SECONDS}s  persist={PERSIST_INTERVAL}s")
    out("=" * 60)

    threading.Thread(target=persistence_loop, daemon=True).start()

    monitors: dict[int, TargetMonitor] = {}
    while True:
        try:
            current = find_targets()
            for pid, name in current.items():
                if pid not in monitors:
                    m = TargetMonitor(pid, name)
                    m.start()
                    monitors[pid] = m
            for pid in list(monitors.keys()):
                if pid not in current:
                    monitors.pop(pid).stop()
        except KeyboardInterrupt:
            break
        except Exception as e:
            out(f"[!] loop error: {e}")
        time.sleep(POLL_INTERVAL)

    for m in monitors.values():
        m.stop()
    out("stopped.")


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        pass
